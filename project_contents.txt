# فرفری — راهنمای راه‌اندازی

سایتی خصوصی با پست (متن، عکس، ویدیو، آهنگ، فایل)، کامنت، واکنش، پروفایل، پیام ناشناس،
«سخنان نغزِ امروز» و تغییر ظاهر سایت بر اساس حالِ مدیر.

## زبان و ساختار
- بک‌اند: **Node.js** (نسخه ۱۸ یا بالاتر)، بدون هیچ وابستگی؛ **نیازی به `npm install` نیست**.
- فرانت‌اند: HTML / CSS / JavaScript داخل پوشه‌ی `public`.
- داده‌ها داخل پوشه‌ی `data` ذخیره می‌شوند (دیتابیس + فایل‌های آپلودی).

## اجرا روی کامپیوتر (برای تست)
```
node server.js
```
بعد در مرورگر: http://localhost:3000

## گذاشتن روی هاست
هاست باید **Node.js** را پشتیبانی کند (VPS، یا هاست‌هایی که «Node.js App» دارند مثل cPanel با گزینه‌ی Setup Node.js App،
لیارا، Render، Railway و …). هاست‌های فقط-PHP کار نمی‌کنند.

1. کل پوشه را آپلود کن (یا زیپ را روی هاست باز کن).
2. فایل شروع: `server.js` — دستور اجرا: `npm start` (یا `node server.js`).
3. متغیرهای محیطی را در پنل هاست تنظیم کن:

| متغیر | معنی | پیش‌فرض |
|---|---|---|
| `ADMIN_USER` | نام کاربری مدیر | `Ferferi` |
| `ADMIN_PASS` | رمز مدیر | `112287878` |
| `PORT` | پورت (هاست معمولاً خودش می‌گذارد) | `3000` |
| `DATA_DIR` | مسیر ذخیره‌ی داده‌ها | `./data` |
| `MAX_UPLOAD_MB` | سقف حجم هر آپلود | `200` |
| `TZ_NAME` | منطقه‌ی زمانی | `Asia/Tehran` |

**مهم:** پوشه‌ی `data` را هرگز پاک نکن و از آن بکاپ بگیر؛ همه‌ی پست‌ها و کاربران آنجاست.
اگر هاستت دیسک پایدار ندارد (مثل بعضی پلن‌های رایگان)، `DATA_DIR` را روی یک دیسک دائمی بگذار.

## دامنه‌ی .com
دامنه را از هر ثبت‌کننده‌ی دامنه بخر، بعد در تنظیمات DNS آن، رکورد `A` را به IP هاست یا
رکورد `CNAME` را به آدرس هاست بزن. بیشتر هاست‌ها گواهی HTTPS رایگان هم می‌دهند؛ حتماً فعالش کن.

## نصب روی گوشی (مثل اپ)
- آیفون: Safari ← دکمه‌ی اشتراک‌گذاری ← Add to Home Screen
- اندروید: منوی مرورگر ← Install app
(یا از داخل سایت: منو ← «نصب روی گوشی»)

## امنیت
- رمز کاربران با scrypt هش می‌شود و رمز مدیر فقط در سرور است، نه در کد سایت.
- فایل‌های آپلودی فقط برای کاربران واردشده قابل دیدن است.
- بعد از اولین اجرا، `ADMIN_PASS` را عوض کن.
- پیام‌های ناشناس بدون هیچ اطلاعاتی از فرستنده ذخیره می‌شوند.

## فونت‌ها
به‌صورت پیش‌فرض از Google Fonts لود می‌شوند. اگر باز نمی‌شود، راهنمای `public/fonts/README.txt` را ببین.

## افزودن جمله‌های عاشقانه
از داخل سایت: مدیریت ← «جمله‌های من». جمله‌های پیش‌فرض در `content.js` هستند.

/**
 * سرور سایت فرفری — Node.js خالص، بدون هیچ وابستگی (npm install لازم نیست)
 * اجرا:  node server.js

const fs = require('fs');
const path = require('path');
const zlib = require('zlib');
const crypto = require('crypto');

/* ---------- تنظیمات (از متغیرهای محیطی) ---------- */

const DATA = process.env.DATA_DIR || path.join(__dirname, 'data');
const UPLOADS = path.join(DATA, 'uploads');

const ADMIN_PASS = process.env.ADMIN_PASS || '112287878';
const TZ = process.env.TZ_NAME || 'Asia/Tehran';
const MAX_UPLOAD = (+process.env.MAX_UPLOAD_MB || 200) * 1048576;
const H12 = 12 * 3600e3;
const REACTIONS = ['❤️', '🌀', '🥹', '😍', '🔥'];

fs.mkdirSync(UPLOADS, { recursive: true });

/* ---------- دیتابیس (فایل JSON با ذخیره‌ی امن) ---------- */
const DBF = path.join(DATA, 'db.json');
const db = { users: [], posts: [], comments: [], anon: [], moods: [], lastQuote: {}, customQuotes: [], files: {}, siteMood: 'neutral', ver: 1 };
try { Object.assign(db, JSON.parse(fs.readFileSync(DBF, 'utf8'))); } catch (e) { /* اولین اجرا */ }
let saveTimer = null;
function flush() { clearTimeout(saveTimer); const tmp = DBF + '.tmp'; fs.writeFileSync(tmp, JSON.stringify(db)); fs.renameSync(tmp, DBF); }
function save() { db.ver++; clearTimeout(saveTimer); saveTimer = setTimeout(flush, 250); }
for (const sig of ['SIGTERM', 'SIGINT']) process.on(sig, () => { try { flush(); } catch (e) { } process.exit(0); });

/* ---------- ابزارهای امنیتی ---------- */
const SECRET = (() => {
  if (process.env.SECRET) return process.env.SECRET;
  const f = path.join(DATA, 'secret.key');
  try { return fs.readFileSync(f, 'utf8'); } catch (e) { const s = crypto.randomBytes(32).toString('hex'); fs.writeFileSync(f, s, { mode: 0o600 }); return s; }
})();
const sign = s => crypto.createHmac('sha256', SECRET).update(s).digest('base64url');
const rid = () => crypto.randomBytes(12).toString('hex');
const hashPw = (pw, salt) => crypto.scryptSync(String(pw), salt, 64).toString('hex');
const safeEq = (a, b) => { const x = Buffer.from(String(a)), y = Buffer.from(String(b)); return x.length === y.length && crypto.timingSafeEqual(x, y); };
function mkToken(id) { const p = Buffer.from(JSON.stringify({ u: id, e: Date.now() + 60 * 864e5 })).toString('base64url'); return p + '.' + sign(p); }
function readToken(t) {
  if (!t) return null; const [p, s] = t.split('.'); if (!p || !s || !safeEq(s, sign(p))) return null;
  try { const o = JSON.parse(Buffer.from(p, 'base64url').toString()); return o.e > Date.now() ? o.u : null; } catch (e) { return null; }
}
const hits = new Map();
function allow(key, max, ms) { const n = Date.now(); const a = (hits.get(key) || []).filter(t => n - t < ms); if (a.length >= max) { hits.set(key, a); return false; } a.push(n); hits.set(key, a); return true; }
setInterval(() => { const n = Date.now(); for (const [k, a] of hits) if (!a.length || n - a[a.length - 1] > 36e5) hits.delete(k); }, 6e5).unref();

/* ---------- ساخت حساب مدیر ---------- */
(function ensureAdmin() {
  let a = db.users.find(u => u.id === 'admin');
  if (!a) { a = { id: 'admin', disp: '', bio: '', avatar: null, created: Date.now() }; db.users.push(a); }
  a.name = ADMIN_USER; a.key = ADMIN_USER.toLowerCase();
  flush();
})();
const RESERVED = () => ['ferferi', 'فرفری', ADMIN_USER.toLowerCase()];

/* ---------- کمکی‌ها ---------- */
const err = (s, m) => Object.assign(new Error(m), { status: s });
const json = (res, code, obj) => { if (res.headersSent) return; const b = JSON.stringify(obj); res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store', 'Content-Length': Buffer.byteLength(b) }); res.end(b); };
const need = c => { if (!c.u) throw err(401, 'ابتدا وارد شو.'); return c.u; };
const adminOnly = c => { const u = need(c); if (u.id !== 'admin') throw err(403, 'این بخش فقط مخصوص مدیر است.'); return u; };
const byId = id => db.users.find(u => u.id === id);
const pub = u => u ? { name: u.name, disp: u.disp || '', bio: u.bio || '', avatar: u.avatar ? '/uploads/' + u.avatar : null, admin: u.id === 'admin' } : { name: '؟', disp: 'کاربر حذف‌شده', bio: '', avatar: null, admin: false };
const clean = (s, n) => String(s == null ? '' : s).replace(/\u0000/g, '').trim().slice(0, n);
const tkey = () => new Date().toLocaleDateString('en-CA', { timeZone: TZ });
const dayNum = () => Math.floor(Date.parse(tkey()) / 864e5);
function readJson(req, max = 200e3) {
  return new Promise((ok, no) => {
    let n = 0; const ch = [];
    req.on('data', c => { n += c.length; if (n > max) { no(err(413, 'حجم درخواست زیاد است.')); req.destroy(); } else ch.push(c); });
    req.on('end', () => { try { ok(JSON.parse(Buffer.concat(ch).toString() || '{}')); } catch (e) { no(err(400, 'درخواست نامعتبر است.')); } });
    req.on('error', no);
  });
}
const cookies = req => Object.fromEntries((req.headers.cookie || '').split(';').map(s => s.trim().split('=')).filter(a => a[0]).map(([k, ...v]) => [k, v.join('=')]));
function setSession(res, req, id) {
  const secure = req.headers['x-forwarded-proto'] === 'https' || req.socket.encrypted;
  res.setHeader('Set-Cookie', `fs=${id ? mkToken(id) : ''}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${id ? 60 * 86400 : 0}${secure ? '; Secure' : ''}`);
}
const ipOf = req => (req.headers['x-forwarded-for'] || '').split(',')[0].trim() || req.socket.remoteAddress || '?';

/* ---------- خروجی پست و کامنت ---------- */
const cOut = (c, forAdmin) => ({ id: c.id, postId: c.postId, user: pub(byId(c.userId)), text: c.text, ts: c.ts, ...(forAdmin ? { seen: !!c.seen } : {}) });
function postOut(p, me) {
  const rx = {}; for (const e of Object.values(p.reactions || {})) rx[e] = (rx[e] || 0) + 1;
  return {
    id: p.id, text: p.text, ts: p.ts, pinned: !!p.pinned, author: pub(byId('admin')),
    media: p.media.map(m => ({ ...m, url: '/uploads/' + m.id })),
    reactions: rx, mine: (p.reactions || {})[me.id] || null,
    comments: db.comments.filter(c => c.postId === p.id).sort((a, b) => a.ts - b.ts).map(c => cOut(c))
  };
}
function removeFile(id) { if (!id || !db.files[id]) return; delete db.files[id]; fs.unlink(path.join(UPLOADS, id), () => { }); }
const unseenCounts = () => ({ comments: db.comments.filter(c => !c.seen && c.userId !== 'admin').length, anon: db.anon.filter(a => !a.seen).length });

/* ---------- مسیرها ---------- */
const routes = [];
const route = (m, re, fn) => routes.push([m, re, fn]);

// --- ورود و ثبت‌نام
route('GET', /^\/api\/site$/, () => ({ mood: db.siteMood }));
route('GET', /^\/api\/me$/, c => ({ user: c.u ? pub(c.u) : null, mood: db.siteMood, ver: db.ver }));
route('POST', /^\/api\/signup$/, async c => {
  if (!allow('su:' + c.ip, 10, 36e5)) throw err(429, 'تعداد تلاش‌ها زیاد بود؛ کمی بعد دوباره امتحان کن.');
  const b = await readJson(c.req), name = clean(b.name, 20), code = clean(b.code, 30);
  if (!/^[\p{L}\p{N} _.\-]{2,20}$/u.test(name)) throw err(400, 'نام باید ۲ تا ۲۰ حرف باشد.');
  if (code.length < 3) throw err(400, 'کد ورود باید حداقل ۳ کاراکتر باشد.');
  if (RESERVED().includes(name.toLowerCase())) throw err(400, 'این نام رزرو شده است؛ نام دیگری انتخاب کن.');
  if (db.users.some(u => u.key === name.toLowerCase())) throw err(400, 'این نام قبلاً گرفته شده است.');
  const salt = crypto.randomBytes(16).toString('hex');
  const u = { id: rid(), name, key: name.toLowerCase(), salt, hash: hashPw(code, salt), disp: '', bio: '', avatar: null, created: Date.now() };
  db.users.push(u); save(); setSession(c.res, c.req, u.id);
  return { user: pub(u), mood: db.siteMood };
});
route('POST', /^\/api\/login$/, async c => {
  if (!allow('li:' + c.ip, 15, 6e5)) throw err(429, 'تلاش‌های ناموفق زیاد بود؛ چند دقیقه بعد دوباره امتحان کن.');
  const b = await readJson(c.req), name = clean(b.name, 20), code = clean(b.code, 30);
  let u = db.users.find(u => u.key === name.toLowerCase());
  let ok = false;
  if (u && u.id === 'admin') ok = safeEq(code, ADMIN_PASS);
  else if (u) ok = safeEq(hashPw(code, u.salt), u.hash);
  if (!ok) throw err(401, 'نام یا کد ورود درست نیست.');
  setSession(c.res, c.req, u.id);
  return { user: pub(u), mood: db.siteMood };
});
route('POST', /^\/api\/logout$/, c => { setSession(c.res, c.req, null); return { ok: true }; });

// --- پروفایل
route('PUT', /^\/api\/me$/, async c => {
  const u = need(c), b = await readJson(c.req);
  const disp = clean(b.disp, 20), bio = clean(b.bio, 160);
  if (disp && u.id !== 'admin' && RESERVED().includes(disp.toLowerCase())) throw err(400, 'این نام نمایشی رزرو شده است.');
  if (b.code) {
    if (u.id === 'admin') throw err(400, 'رمز مدیر از تنظیمات سرور عوض می‌شود.');
    const code = clean(b.code, 30); if (code.length < 3) throw err(400, 'کد جدید باید حداقل ۳ کاراکتر باشد.');
    u.salt = crypto.randomBytes(16).toString('hex'); u.hash = hashPw(code, u.salt);
  }
  if (b.avatar === null) { removeFile(u.avatar); u.avatar = null; }
  else if (b.avatar && b.avatar !== u.avatar) {
    const f = db.files[b.avatar]; if (!f || f.owner !== u.id || f.type !== 'image') throw err(400, 'عکس نامعتبر است.');
    removeFile(u.avatar); u.avatar = b.avatar;
  }
  u.disp = disp; u.bio = bio; save();
  return { user: pub(u) };
});

/* ---------- ارسال فایل با پشتیبانی Range (لازم برای ویدیو و آهنگ در سافاری) ---------- */
function sendFile(req, res, fp, mime, o = {}) {
  let st; try { st = fs.statSync(fp); } catch (e) { return json(res, 404, { error: 'پیدا نشد.' }); }
  const h = { 'Content-Type': mime, 'Accept-Ranges': 'bytes', 'Cache-Control': o.cache || 'private, max-age=86400' };
  if (o.download) h['Content-Disposition'] = `attachment; filename*=UTF-8''${encodeURIComponent(o.name || 'file')}`;
  const m = /^bytes=(\d*)-(\d*)$/.exec(req.headers.range || '');
  if (m && (m[1] || m[2])) {
    let s = m[1] ? +m[1] : Math.max(0, st.size - +m[2]), e = m[1] && m[2] ? +m[2] : st.size - 1;
    if (e >= st.size) e = st.size - 1;
    if (s > e || s >= st.size) { res.writeHead(416, { 'Content-Range': `bytes */${st.size}` }); return res.end(); }
    h['Content-Range'] = `bytes ${s}-${e}/${st.size}`; h['Content-Length'] = e - s + 1; res.writeHead(206, h);
    if (req.method === 'HEAD') return res.end();
    return fs.createReadStream(fp, { start: s, end: e }).on('error', () => res.end()).pipe(res);
  }
  h['Content-Length'] = st.size; res.writeHead(200, h);
  if (req.method === 'HEAD') return res.end();
  fs.createReadStream(fp).on('error', () => res.end()).pipe(res);
}

const MIME = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.json': 'application/json', '.webmanifest': 'application/manifest+json', '.svg': 'image/svg+xml', '.png': 'image/png', '.ico': 'image/x-icon', '.woff2': 'font/woff2', '.woff': 'font/woff', '.ttf': 'font/ttf' };
function serveStatic(req, res, p) {
  let rel; try { rel = decodeURIComponent(p); } catch (e) { return json(res, 400, { error: 'آدرس نامعتبر' }); }
  if (rel.endsWith('/')) rel += 'index.html';
  let f = path.join(PUBLIC, path.normalize(rel));
  if (!f.startsWith(PUBLIC)) return json(res, 403, { error: 'ممنوع' });
  let st; try { st = fs.statSync(f); } catch (e) { st = null; }
  if (!st || st.isDirectory()) { if (path.extname(rel)) return json(res, 404, { error: 'پیدا نشد.' }); f = path.join(PUBLIC, 'index.html'); try { st = fs.statSync(f); } catch (e) { return json(res, 404, { error: 'index.html پیدا نشد.' }); } }
  const ext = path.extname(f), type = MIME[ext] || 'application/octet-stream', etag = `W/"${st.size}-${Math.round(st.mtimeMs)}"`;
  if (req.headers['if-none-match'] === etag) { res.writeHead(304); return res.end(); }
  const long = ['.woff2', '.woff', '.ttf', '.png'].includes(ext);
  const h = { 'Content-Type': type, ETag: etag, 'Cache-Control': long ? 'public, max-age=604800' : 'no-cache' };
  if (/text|javascript|json|svg|manifest/.test(type) && /gzip/.test(req.headers['accept-encoding'] || '')) {
    h['Content-Encoding'] = 'gzip'; h.Vary = 'Accept-Encoding'; res.writeHead(200, h);
    return req.method === 'HEAD' ? res.end() : fs.createReadStream(f).pipe(zlib.createGzip()).pipe(res);
  }
  h['Content-Length'] = st.size; res.writeHead(200, h);
  if (req.method === 'HEAD') return res.end(); fs.createReadStream(f).pipe(res);
}

    res.setHeader('X-Content-Type-Options', 'nosniff'); res.setHeader('Referrer-Policy', 'same-origin'); res.setHeader('X-Frame-Options', 'DENY');
    const uid = readToken(cookies(req).fs), u = uid ? byId(uid) : null;
    const ctx = { req, res, url, u: u || null, ip: ipOf(req) };

    if (p === '/healthz') return json(res, 200, { ok: true });
    if (p.startsWith('/api/')) {
      for (const [m, re, fn] of routes) {
        if (m !== req.method) continue; const mm = re.exec(p); if (!mm) continue;
        const out = await fn(ctx, ...mm.slice(1)); return json(res, 200, out === undefined ? { ok: true } : out);
      }
      return json(res, 404, { error: 'مسیر پیدا نشد.' });
    }
    if (p.startsWith('/uploads/')) {
      if (!u) return json(res, 401, { error: 'ابتدا وارد شو.' });
      const id = p.slice(9), f = db.files[id]; if (!/^[a-f0-9]{24}$/.test(id) || !f) return json(res, 404, { error: 'پیدا نشد.' });
      const media = f.type === 'image' || f.type === 'video' || f.type === 'audio';
      return sendFile(req, res, path.join(UPLOADS, id), media ? f.mime : 'application/octet-stream', { download: !media || url.searchParams.has('dl'), name: f.name });
    }
    if (req.method !== 'GET' && req.method !== 'HEAD') return json(res, 405, { error: 'روش نامعتبر' });
    serveStatic(req, res, p);
  } catch (e) {
    if (e && e.status) json(res, e.status, { error: e.message });
    else { console.error(e); json(res, 500, { error: 'خطای سرور؛ دوباره امتحان کن.' }); }
  }
});

async function shrink(f) {
  if (f.type === 'image/gif') return f;
  try { const b = await createImageBitmap(f), s = Math.min(1, 1800 / Math.max(b.width, b.height)), c = document.createElement('canvas'); c.width = Math.round(b.width * s); c.height = Math.round(b.height * s); c.getContext('2d').drawImage(b, 0, 0, c.width, c.height); return await new Promise(r => c.toBlob(r, 'image/jpeg', .86)) || f; } catch (e) { return f; }
}
async function addFiles(e) {
  const files = [...e.target.files]; e.target.value = '';
  for (const f of files) {
    if (att.length >= 8) { toast('حداکثر ۸ فایل'); break; }
    const type = f.type.startsWith('image/') ? 'image' : f.type.startsWith('video/') ? 'video' : f.type.startsWith('audio/') ? 'audio' : 'file';
    const a = { type, name: f.name, pct: 0, id: null, preview: type === 'image' ? URL.createObjectURL(f) : null }; att.push(a); drawAtt();
    (async () => {
      try { const blob = type === 'image' ? await shrink(f) : f; const nm = type === 'image' && blob !== f ? f.name.replace(/\.[^.]+$/, '') + '.jpg' : f.name; const d = await upload(blob, nm, 'post', p => { a.pct = p; drawAtt(); }); a.id = d.id; a.pct = 1; }
      catch (x) { toast(x.message); att = att.filter(y => y !== a); } drawAtt();
    })();
  }
}
